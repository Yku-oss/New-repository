#include<iostream>
#include<string>
#include<vector>
#include<algorithm>
#include<deque>
using namespace std;
using ll = long long;

void solve(){
	int n , k;
	cin >> n >> k;
	vector<ll>a(n);
	for(ll i = 0; i < n ;i++)cin >> a[i];

	vector<int>ma;//储存最大值 
	
    vector<int>mi;//储存最小值 
	
    deque<int> dp ;//储存最大值/最小值下标 
	
    for(int i = 0 ; i < n ;i++){
		if(!dp.empty() && dp.front() <= i - k){
			dp.pop_front();//弹出队头，因为队头不在窗口中的时候就是无效的 
		}
		
        while(!dp.empty() && a[dp.back()] >= a[i]) dp.pop_back();// 遇到当前最小值就把队列清空 
		
        dp.push_back(i);//入队 
		
        if( i >= k - 1)mi.push_back(a[dp.front()]); // 当满足窗口时，就把最大/最小值放入 
	}
	
	dp.clear(); //找出最大值 
	
    for(int i = 0 ; i < n ;i++){
		
        if(!dp.empty() && dp.front() <= i - k){
			dp.pop_front();
		}
		
        while(!dp.empty() && a[dp.back()] <= a[i]) dp.pop_back();
		
        dp.push_back(i);
		
        if( i >= k - 1)ma.push_back(a[dp.front()]);
	}
	
	
	int m = mi.size();
	
    for(int i = 0 ; i < m ;i++){
		cout << mi[i] <<" ";
	}
	
    cout << "\n";
	
    for(int i = 0 ; i < m ;i++){
		cout << ma[i] <<" ";
	}
	
	}


int main(){
	ios::sync_with_stdio(false);
	cout.tie(0);
	cin.tie(0);
	int t = 1;
	while(t--){
		solve();
	}
}
