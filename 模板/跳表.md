
## 一、跳表是什么

跳表是一种**概率平衡**的数据结构，功能上可以替代平衡树（AVL/红黑树），但代码更简单，实现容易。

**核心思想**：通过多层链表 + 随机化，实现 O(log n) 的查找、插入、删除。

## 二、完整代码（可直接用）

```cpp
#include <iostream>
#include <vector>
#include <cstdlib>
#include <climits>
using namespace std;


// 跳表实现
// k 为键值 ， v为对应值的类型
template<typename K, typename V>
class SkipList {
private:
    struct Node {
        K key;   // 用于后面对值的查找
        V value; // 储存key 的值
        vector<Node*> next; //指向下一节点的指针数组，长度为该节点的层数 ， 比如 next[i] 指的是i层级的下一个节点。

        // 构造函数：创建一个新节点
        // 参数：k - 键，v - 值，level - 节点的层数（从0开始）

        Node(K k, V v, int level) : key(k), value(v), next(level + 1, nullptr) {}
    };

    int maxLevel;  //表示最大层数，通常设置为32或64
    double probability; // 层数增加后的概率
    Node* head;  // 跳表的头节点，所有层的起始节点
    int size;   // 当前跳表中元素的数量

    // 随机生成层数
    int randomLevel() {
        // 生成随机数，以 probability 的概率增加层数，不超过 maxLevel
        int level = 0;
        while ((rand() % 100) < probability * 100 && level < maxLevel) {
            level++;
        }
        return level;
    }

public:
    SkipList(int maxLvl = 32, double p = 0.25) : maxLevel(maxLvl), probability(p), size(0) {
        // 创建头节点，键设为最小值，值无意义，层数为最大层数
        head = new Node(INT_MIN, INT_MIN, maxLevel);
        srand(time(nullptr)); // 初始化随机数生成器
    }

    // 插入或更新
    void put(K key, V value) {
        // update数组：记录每一层需要更新的前驱节点
        // update[i] 表示在第 i 层，新节点应该插入在 update[i] 之后
        vector<Node*> update(maxLevel + 1, nullptr);
        Node* cur = head; 

        // 1. 从上往下找，记录每层需要更新的前驱节点
        // 就是从最高层往下寻找，直到找到合适的位置插入新节点
        for (int i = maxLevel; i >= 0; i--) {
             // 在当前层向右移动，直到下一个节点的键 >= 目标键
            while (cur->next[i] && cur->next[i]->key < key) {
                cur = cur->next[i];
            }
            // 记录并更新当前层要插入的位置的前驱节点
            update[i] = cur;
        }
         // 此时 cur 在第0层，且 cur->next[0] 是第一个键 >= key 的节点


        // 2. 检查 key 是否已存在
        // 如果存在，则更新其值
        if (cur->next[0] && cur->next[0]->key == key) {
            cur->next[0]->value = value;
            return;
        }


        // 3. 生成随机层数，创建新节点
        int newLevel = randomLevel();
        Node* newNode = new Node(key, value, newLevel);


        // 4. 在每一层插入新节点
        // 只需要在 0 到 新节点newLevel 层插入
        for (int i = 0; i <= newLevel; i++) {
            newNode->next[i] = update[i]->next[i];
            update[i]->next[i] = newNode;
        }
        size++;
    }

    // 删除
    void erase(K key) {

        vector<Node*> update(maxLevel + 1, nullptr);
        Node* cur = head;
       // 从上往下找，记录每层需要更新的前驱节点，与上一致 
        for (int i = maxLevel; i >= 0; i--) {
            while (cur->next[i] && cur->next[i]->key < key) {
                cur = cur->next[i];
            }
            update[i] = cur;
        }

        //检查是否存在要删除的节点
        if (cur->next[0] && cur->next[0]->key == key) {
            Node* toDelete = cur->next[0];
            for (int i = 0; i <= maxLevel; i++) {
                  // 如果当前层的前驱节点的下一个节点是要删除的节点，就把它指向要删除节点的下一个节点
                if (update[i]->next[i] == toDelete) {
                    update[i]->next[i] = toDelete->next[i];
                }
            }
            delete toDelete;//释放内存
            size--;
        }
    }

    // 查找
    V get(K key) {
        Node* cur = head;
        for (int i = maxLevel; i >= 0; i--) {
            while (cur->next[i] && cur->next[i]->key < key) {
                cur = cur->next[i];
            }
        }
        if (cur->next[0] && cur->next[0]->key == key) {
            return cur->next[0]->value;
        }
        return V();
    }

    // 是否存在
    bool contains(K key) {
        Node* cur = head;
        for (int i = maxLevel; i >= 0; i--) {
            while (cur->next[i] && cur->next[i]->key < key) {
                cur = cur->next[i];
            }
        }
        return cur->next[0] && cur->next[0]->key == key;
    }

    // // 打印跳表结构（调试用）
    // void print() {

    //         cout << "Level " << i << ": ";
    //         Node* cur = head->next[i];
    //         while (cur) {
    //             cout << cur->key << "(" << cur->value << ") → ";
    //             cur = cur->next[i];
    //         }
    //         cout << "null" << endl;
    //     }
    // }

    // int getSize() { return size; }
};

int main() {
    SkipList<int, string> skipList;
    skipList.put(1, "one");
    skipList.put(3, "three");
    skipList.put(2, "two");
    skipList.put(5, "five");
    skipList.put(4, "four");

    cout << "Skip List after insertions:" << endl;
    skipList.print();

    cout << "\nGet key 3: " << skipList.get(3) << endl;
    cout << "Contains key 2: " << (skipList.contains(2) ? "Yes" : "No") << endl;

    skipList.erase(3);
    cout << "\nSkip List after deleting key 3:" << endl;
    skipList.print();

    return 0;
}
```

````
对比项	        跳表	        AVL/红黑树
实现难度	    简单	            复杂
插入/删除	期望 O(log n)	    O(log n)
有序遍历	    方便	            方便
空间占用	    略大（多层指针）	    较小
实际应用	    Redis Zset	        C++ map/set
````
